private func handleVideoCallTap() async {
    // 1. peerUserId nil
    guard let peer = peerUserId else {
        toast(L10n.chatCallPeerOffline)
        return
    }

    // 2. 5s camera cooldown（S9 spike 确认起点：直播开始 or 直播结束）
    let cooldownBase = LiveStore.shared.liveStartTime ?? Date.distantPast
    let elapsed = Date().timeIntervalSince(cooldownBase)
    guard elapsed >= 5 else {
        toast(L10n.chatCameraBusy)
        return
    }

    // 3. 对方在线状态（S8 spike 确认字段）
    let status = await UserProfileService.shared.fetchOnlineStatus(userId: peer)

    case .online: break
    case .busy:
        toast(L10n.chatCallUserBusy)
        return
    case .offline:
        toast(L10n.chatCallUserOffline)
        return
    }

    // 4. CallStore.idle
    guard CallStore.shared.state == .idle else {
        toast(L10n.chatCallInProgress)
        return
    }

    // 5. 发起
    CallStore.shared.callOut(peerYxAccId: peer, callType: .video)
}
