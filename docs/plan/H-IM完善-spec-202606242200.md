### §2.2 MessageRouter protocol（核心抽象）

```swift
/// 消息分发抽象层；多个 router 通过 NIMService 串联，各自挑感兴趣的 attachType。
@MainActor
protocol MessageRouter: AnyObject {
    /// 处理一条消息。返回 true 表示已消费（不再传给下游 router）。
    func route(_ attachType: AttachType, payload: [String: Any], context: MessageContext) -> Bool
}

enum MessageContext {
    case liveChatroom(roomId: String)
    case partyChatroom(roomId: String)
    case p2p(fromAccount: String)
    case sysMsg
    case syncSysMsg   // 离线/漫游补发
}
```

**优先级与短路**：
- NIMService 维护 `[MessageRouter]` 顺序数组
- 收到消息后从前到后依次调 `route`；第一个返回 `true` 的 router 消费后链路终止
- **顺序约定**：PKNIMRouter（G）→ PartyMessageRouter → GiftMessageRouter（礼物会话）→ SystemMessageRouter → 兜底 fallback

## §3 三通道路由实现

### §3.1 SystemMessageRouter（本会话实现）

负责 `nim.on('sysMsg')` 收到的所有 attachType 分发到对应业务 store。

```swift
@MainActor
final class SystemMessageRouter: MessageRouter {
    weak var liveStore: LiveStore?
    weak var callStore: CallStore?
    weak var sessionStore: SessionStore?
    
    func route(_ attachType: AttachType, payload: [String: Any], context: MessageContext) -> Bool {
        guard case .sysMsg = context else { return false }
        switch attachType {
        case .forceEndLive:    liveStore?.handleForceEndFromIM(reason: .disconnected)
        case .banned:          liveStore?.handleForceEndFromIM(reason: .banned)
        case .complianceWarning:
            liveStore?.showComplianceWarning(text: ...)
        case .callRemoteMessage:
            let decoded = decodeRemoteText(payload)
            callStore?.appendRemoteText(decoded)
        case .callPayWaitState:
            callStore?.updateWaitState(type: payload["type"] as? Int ?? 0)
        case .callIncomePerMinute:
            callStore?.appendCallIncome(num: payload["num"] as? Int ?? 0)
        case .callGiftIncome:
            callStore?.appendGiftIncome(num: payload["num"] as? Int ?? 0)
        case .callRechargeReward:
            callStore?.appendWaitBonus(num: payload["giveDiamondNum"] as? Int ?? 0)
        case .followIncrement:
            sessionStore?.incrementFollow()
        case .anchorAuditChange:
            // applyStatus=0 通过→logOut / 1 拒绝→alert
            sessionStore?.handleAuditStatus(payload)
        default:
            return false
        }
        return true
    }
}
```

### §3.3 PartyMessageRouter（本会话从 PartyRoomChatManager 抽出）

@MainActor
final class PartyMessageRouter: MessageRouter {
    weak var partyStore: PartyStore?
    
    func route(_ attachType: AttachType, payload: [String: Any], context: MessageContext) -> Bool {
        guard case .partyChatroom = context else { return false }

        switch attachType {
        case .partyMemberEnter:    partyStore?.handleMemberEnter(payload)

            return false
        }
        return true
    }
}

## §4 NIMService 内部 helper

### §4.1 unwrapDataField（从 PartyRoomChatManager 抽出）

```swift
/// 三态字典 / JSON 字符串 / Base64+gzip 解出业务数据
/// 派对房 2049 / 部分 chatroom 消息会用 gzip 压缩
static func unwrapDataField(_ raw: Any?) -> [String: Any]? {
    // 1) 已是字典
    if let dict = raw as? [String: Any] { return dict }
    // 2) JSON 字符串
    if let s = raw as? String,
       let data = s.data(using: .utf8),
       let dict = try? JSONSerialization.jsonObject(with: data) as? [String: Any] {
        return dict
    }
    // 3) Base64 + gzip
    if let s = raw as? String,
       let compressed = Data(base64Encoded: s),
       let decompressed = try? compressed.gunzipped(),
       let dict = try? JSONSerialization.jsonObject(with: decompressed) as? [String: Any] {
        return dict
    }
    return nil
}
```

@MainActor
private func handleSyncSysMsgs(_ msgs: [NIMSystemNotification]) async {
    // 同步可能一次几百条 → 批 50 / 50ms 防 main actor 阻塞
    for chunk in msgs.chunked(into: 50) {
        for m in chunk {
            await dispatchSysMsg(m, context: .syncSysMsg)
        }
        try? await Task.sleep(nanoseconds: 50_000_000)
    }
}
