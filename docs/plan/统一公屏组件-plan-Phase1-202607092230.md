> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立 `Sources/PublicChat/` 新模块，实装 14 变体 Row + Live adapter，将 `LiveRoomView` 内公屏切换到新组件；删除 `Sources/Live/PublicScreen/UI/ChatRowSubviews.swift`(635 行) 与 `Sources/Live/PublicScreen/PublicChatMessageType.swift`(77 行) 旧实现。Phase 2（Call）与 Phase 3（Party）待 Phase 1 落地后各自 spec-to-plan。

```swift
import Foundation
import Combine

@MainActor
final class UnifiedPublicChatFeed: ObservableObject {
    @Published private(set) var messages: [UnifiedPublicChatMessage] = []
    let limit: Int

    init(limit: Int = 200) {
        self.limit = limit
    }

    func append(_ msg: UnifiedPublicChatMessage) {
        messages.append(msg)
        trimIfNeeded()
    }

    func appendBatch(_ msgs: [UnifiedPublicChatMessage]) {
        messages.append(contentsOf: msgs)
        trimIfNeeded()
    }

    /// 全量替换语义（供 upstream 全量 map 后灌入用；Live 场景 T10 使用）
    func replace(_ msgs: [UnifiedPublicChatMessage]) {
        messages = msgs
        trimIfNeeded()
    }

    func clear() {
        messages.removeAll()
    }

    private func trimIfNeeded() {
        if messages.count > limit {
            messages.removeFirst(messages.count - limit)
        }
    }
}
```

```swift
import Foundation

/// H5 `useUserLevelHooks` + `setLevelStyle` 对齐：
/// - tier = level / 10 向下取整
/// - 若 tier >= 10 → 截断为 10（最高档）
/// - 若 level < 0 → 0（防御）
enum LevelTierResolver {
    static func tier(for level: Int) -> Int {
        guard level >= 0 else { return 0 }
        let raw = level / 10
        return min(raw, 10)
    }
}
```

```swift
import SwiftUI

/// H5 messageScroller `setLevelBg(item)` 对齐：Lv.N 徽章 h14 w39 rounded-27
/// 分档背景从 tier 0-10 派色（先用占位色；未来 K 期 UserLevelBadge 有权威 tier→色映射，可迁）
struct LevelBadge: View {
    let level: Int

    var body: some View {
        HStack(spacing: 2) {
            // 星标（K 期已有 icon_level_x.png / 未来 tier 色，先用系统 star.fill 占位）
            Image(systemName: "star.fill")
                .font(.system(size: 8))
                .foregroundColor(.white)
            Text("Lv.\(level)")
                .font(.system(size: 8, weight: .medium))
                .foregroundColor(.white)
        }
        .frame(width: 39, height: 14)
        .background(tierBackground, in: Capsule())
    }

    private var tierBackground: LinearGradient {
        // Phase 1 用简化 tier→色；未来接 UserLevelBadge/tier gradient 表
```

```swift
import SwiftUI

struct PublicChatRow: View {
    let message: UnifiedPublicChatMessage
    let theme: PublicChatTheme

    var body: some View {

            case .text, .anchor, .gift, .luckyGift, .system:

            case .enterRoom, .officialBoostEnter, .announcement, .pkNotify:

                activityGroup
            case .gameWinNotify, .partyModeSwitch, .bonus:
                EmptyView()   // Phase 2/3 才 emit；Phase 1 不出现

        }

        case .text(let c, let mentions, let translation, let replyToNick):

        case .anchor(let c, let translation):
            RowAnchor(sender: message.sender, content: c, translation: translation, theme: theme)
        case .gift(let url, let name, let count):
            RowGift(sender: message.sender, iconURL: url, name: name, count: count, theme: theme)
        case .luckyGift(let url, let count, let total):
            RowLuckyGift(sender: message.sender, iconURL: url, count: count, totalReward: total, theme: theme)
        case .system(let text):

        }
```

```swift
import Foundation

enum LivePublicChatAdapter {

    static func adapt(_ m: PublicChatMessage) -> UnifiedPublicChatMessage {
        // 系统消息路径
        if m.isSystem {
            return UnifiedPublicChatMessage(
                id: m.id, timestamp: Date(),
                sender: nil,
                variant: .system(text: m.text)
            )
        }
        // 用户消息路径
        let sender = SenderProfile(
            userId: nil,
            nickname: m.senderNickname ?? "",
            avatarURL: m.senderAvatar,
            userLevel: m.userLevel,
            isVip: m.isVip,
            isHost: m.isHost,
            role: nil,
            medals: [],
            chatBubble: nil,
            isPlatformAdmin: false,
            isSelf: false,
            isNewUser: false,
            nicknameColor: m.isHost ? .anchor : .default
        )
        let variant = mapVariant(m)
        return UnifiedPublicChatMessage(id: m.id, timestamp: Date(), sender: sender, variant: variant)
    }

    private static func mapVariant(_ m: PublicChatMessage) -> PublicChatVariant {
        guard let payload = m.rawPayload else {
            // 老构造路径兜底：把 text 当 system 显示（防显示错误 UI）
            return .system(text: m.text)
        }
        switch payload {
        case .text(let s):
            return .text(content: s)
        case .anchorText(let s):
            return .anchor(content: s)
        case .gift(let url, let name, let count):
            return .gift(iconURL: url, name: name, count: count)
        case .luckyGift(let url, let count, let total):
            return .luckyGift(iconURL: url, count: count, totalReward: total)
        case .enterRoom(let vehicle, let item, let boost):
            return boost ? .officialBoostEnter : .enterRoom(vehicleImg: vehicle, itemSmallImg: item)
        case .pkNotify(let html):
            return .pkNotify(richText: PKNotifyRichTextParser.parse(html))
        case .rpsWin(let url, let hours):
            return .rpsWin(medalUrl: url, medalHours: hours)
        case .wheelRes(let text, let hl):
            return .wheelRes(resultText: text, resultHighlight: hl)
        case .announcement(let text):
            return .announcement(text: text, kind: .liveOfficial)
        case .winnerBroadcast(let name, let qty, let img, let cta, let avatar):
            return .winnerBroadcast(activityName: name, quantity: qty, imageURL: img, joinCTA: cta, avatar: avatar)
        case .wishlistEffect(let text, let icon):
            return .wishlistEffect(text: text, iconURL: icon)
        case .diamondGift(let sub):
            return .diamondGift(subType: sub)
        case .system(let s):
            return .system(text: s)
        }
    }
}

/// 简版 PK notify 富文本解析（H5 v-html 拼接的 span+img 混排）。
/// Phase 1 先按 plain text 兜底显示；Phase 1 impl 时按 H5 真实 attachType=-9 payload 结构补 iconURL/highlight 分段。
enum PKNotifyRichTextParser {
    static func parse(_ html: String) -> [RichSegment] {
        // 兜底：去掉 HTML tags 后作为一段 white 文本
        let plain = html.replacingOccurrences(of: "<[^>]+>", with: "", options: .regularExpression)
        return [.text(plain, color: .white)]
    }
}
```
