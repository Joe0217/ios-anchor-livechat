### 1.2 CallStore 新增字段

```swift
// 充值等待锁定态视觉倒计时（60→0 每秒 -1；PAY_SUCCESS 清；hangup timer 归 0）
@Published private(set) var callWaitCountdown: Int = 0
// PAY_SUCCESS 弹窗触发 token（UUID nil→UUID→nil；view .task(id:) 消费展示 3s 自消）
@Published private(set) var congratsBonusToken: UUID?
// 当次充值成功 bonus 数（PAY_SUCCESS 触发弹窗时读）
@Published private(set) var lastCongratsBonus: Int = 0
// waitState 是否处于锁定态（type=1/3 期间 true；用于 CallHudOverlay 隐藏普通 pill + 显示 WaitRechargeTips）
var isCallWaitLocked: Bool { callWaitState == 1 || callWaitState == 3 }

// 内部
private var callWaitTimerTask: Task<Void, Never>?
private var callWaitPauseAt: Date?

private static let callWaitPrimarySeconds = 60
private static let callWaitFallbackSeconds = 5
private static let congratsBonusDelaySeconds: TimeInterval = 2
private static let congratsBonusVisibleSeconds: TimeInterval = 3
```

### 1.3 updateWaitState 扩展

```swift
func updateWaitState(type: Int) {
    let prev = callWaitState
    callWaitState = type
    AppLogger.call.info("[CallStore] updateWaitState type=\(type) prev=\(prev)")
    switch type {
    case 1, 3: startCallWaitLockTimer(reason: type)      // START_PAY / CALL_TIME_END
    case 2:    handlePaySuccess()                         // PAY_SUCCESS
    case 4:    cancelCallWaitLockTimer()                  // PAY_CANCEL
    case 0:    cancelCallWaitLockTimer()                  // 兜底清
    default:   break
    }
}
```

### 1.4 状态机方法

```swift
/// 启动 60s 主段 + 5s 兜底段自动挂断（H5 topBar.vue callWaitTime + 5s fallback）。
/// 幂等：重复 START_PAY / CALL_TIME_END cancel 旧 task 重启。
/// callWaitPauseAt 记录暂停时刻供 PAY_SUCCESS 补偿。
private func startCallWaitLockTimer(reason: Int) {
    callWaitPauseAt = Date()
    callWaitTimerTask?.cancel()
    callWaitCountdown = Self.callWaitPrimarySeconds
    callWaitTimerTask = Task { @MainActor [weak self] in
        guard let self else { return }
        // 主段 60s 倒计时
        for _ in 0..<Self.callWaitPrimarySeconds {
            try? await Task.sleep(nanoseconds: 1_000_000_000)
            if Task.isCancelled { return }
            self.callWaitCountdown = max(0, self.callWaitCountdown - 1)
        }
        // 兜底段 5s（H5：60s 后再 5s 等 PAY_SUCCESS）
        try? await Task.sleep(nanoseconds: UInt64(Self.callWaitFallbackSeconds) * 1_000_000_000)
        if Task.isCancelled { return }
        // 到期自动挂断
        AppLogger.call.notice("⏰ [CallStore] callWaitLockTimer 兜底到期 → hangup")
        await self.hangup()
    }
}

/// PAY_SUCCESS：清 timer + 补偿 callElapsed + 触发 Congrats 弹窗。
private func handlePaySuccess() {
    // 补偿：callElapsed 已含暂停期，H5 语义是 callingStartTime += pauseDuration
    // iOS 等价：callElapsed 减去暂停时长（让"通话有效时长"不含充值等待）
    if let pauseAt = callWaitPauseAt {
        let paused = Int(Date().timeIntervalSince(pauseAt))
        callElapsed = max(0, callElapsed - paused)
        AppLogger.call.info("[CallStore] paySuccess 补偿 -\(paused)s → callElapsed=\(self.callElapsed)")
    }
    cancelCallWaitLockTimer()
    // 2s 延迟弹 Congrats（H5 showGiftDialog 2s 后）
    let bonus = callWaitBonus  // 快照本次奖励值
    lastCongratsBonus = bonus
    Task { @MainActor [weak self] in
        try? await Task.sleep(nanoseconds: UInt64(Self.congratsBonusDelaySeconds * 1_000_000_000))
        guard let self else { return }
        self.congratsBonusToken = UUID()
    }
}

/// 清 timer + 归 0（PAY_CANCEL / endLocally / hangup 前）。幂等。
private func cancelCallWaitLockTimer() {
    callWaitTimerTask?.cancel()
    callWaitTimerTask = nil
    callWaitCountdown = 0
    callWaitPauseAt = nil
}
```

## §3 验收清单

### F 正向
- F1 · sysMsg -6 type=1（START_PAY）→ CallWaitRechargeTips 顶部滑入 + 60s 倒计时递减 + CallHudOverlay 普通 pill 隐藏
- F2 · sysMsg -6 type=2（PAY_SUCCESS）+ sysMsg 90 bonus=50 → tips 消失 + callElapsed 减去暂停时长 + 2s 后弹 Congrats sheet "grab your 50 💎 bonus"
- F3 · sysMsg -6 type=3（CALL_TIME_END）→ 同 F1 启动 60s 倒计时
- F4 · sysMsg -6 type=4（PAY_CANCEL）→ tips 消失 + timer 清 + 通话继续（不 hangup）
- F5 · 60s + 5s 兜底段结束仍未 type=2 → 自动 hangup(.localHangUp)
- F6 · Congrats sheet 点 OK / drag dismiss → congratsBonusToken 回 nil

### R 反向
- R1 · 通话结束前 timer 未启动 → PAY_SUCCESS 到达不崩溃（callWaitPauseAt=nil guard）
- R2 · endLocally 时 timer 挂着 → cancel + reset（不异步触发 hangup）
- R3 · 重复 type=1 → 幂等 cancel 旧 task 重启（不叠加）
- R4 · alert 打开时 tips 隐藏（confirmationDialog 层级更高，视觉自动挡）
- R5 · state != .connected 时 updateWaitState 不启动 timer（应加 state guard 防抢跑）
