#### T6 — CallStore.acceptIncomingFromLive(msg:) 自动接听入口（0.3d）

**文件**：`Sources/Call/CallStore.swift`

**改动点**：
1. 公开方法：
   ```swift
   func acceptIncomingFromLive(msg: CallMessage) async {
       guard state == .idle, let signaling else { return }
       guard let fromRoomId = msg.fromRoomId, !fromRoomId.isEmpty else { return }
       
       // 1) 初始化 currentCallInfo（被叫 in / frontGameType=.live）
       var info = CurrentCallInfo()
       info.frontGameType = .live
       info.inOrOut = .in
       info.channelId = fromRoomId
       info.callId = msg.callId
       info.remoteUserId = msg.fromUserId
       info.callStartTime = Date().timeIntervalSince1970 * 1000
       current = info
       state = .calling
       
       // 2) 立刻发 Accept（无 UI 确认）
       let ok = await signaling.publish(buildMessage(action: .accept))
       guard ok else {
           lastError = "私 call 发送接听失败"
           await endLocally(reason: .beginCallError, rateCategory: nil, rateType: .callee, abnormal: 1)
           return
       }
       state = .connecting
       
       // 3) join RTC（通话频道）
       await joinRtc(channel: fromRoomId, rateType: .callee)
       
       // 4) 远端可能已在频道，手动补升级
       if agora.remoteUid != 0, state == .connecting {
           current.callConnectTime = Date().timeIntervalSince1970 * 1000
           state = .connected
       }
       
       // 5) joinCall 拉对方资料（异步，UI 后填充）
       Task { @MainActor in
           do {
               let r = try await CallService.joinCall(channelId: fromRoomId)
               guard self.state != .idle, self.current.callId == msg.callId else { return }
               self.current.remoteYxAccid = r.yxAccid ?? self.current.remoteYxAccid
               self.current.remoteNickname = r.nickname ?? self.current.remoteNickname
               self.current.remoteIcon = r.icon ?? self.current.remoteIcon
           } catch {
               print("⚠️ [CallStore] LIVE joinCall 失败 err=\(error.localizedDescription)")
           }
       }
   }
   ```

### M2 — LiveStore 暂停-恢复主流程（1.2 人/天）

> 目标：LiveStore 完整实现 pauseForCall + resumeCall 链路；通过 CallStoreObserver 桥接 CallStore.state 变化触发 resumeCall。
> **依赖**：B 里程碑 HeartbeatController 必须暴露 pause/resume 方法（开工前确认）。

#### T2 — LiveStore.pauseForCall(msg:)（0.4d）

**文件**：`Sources/Live/LiveStore.swift`

**改动点**：

```swift
@MainActor
extension LiveStore {
    /// 直播态收到 RTM VideoCall → 暂停直播 → 自动接听。
    /// 严格对齐 H5 useCallApi.callInEnter:
    ///   if (appStore.isLiving) await livingEnd(false)  // 暂停不清直播信息
    ///   await callApi.accept(callUserId)
    func pauseForCall(msg: CallMessage) async {
        guard state == .living, callState == 0 else {
            print("⚠️ [LiveStore] pauseForCall 跳过 state=\(state) callState=\(callState)")
            return
        }
        
        print("🟡 [LiveStore] pauseForCall 开始：暂停直播 → 接听 \(msg.fromUserId)")
        
        // 1) 停心跳（复用 B 里程碑 HeartbeatController.stop，自动重置 failureCount=0）
        heartbeat.stop()
        
        // 2) 停网络监控（复用 B 里程碑 NetworkQualityMonitor.stop）
        monitor.stop()
        
        // 3) RTC leave 直播频道（不清 roomId/agoraChannelId/rtcToken）
        liveAgora.leave()
        
        // 4) 状态切换
        callState = 1
        
        // 5) WS 在线态联动
        WSHeartbeat.shared.notifyCallStateChanged(callState: 1)
        
        // 6) 委托 CallStore 自动接听
        await CallStore.shared.acceptIncomingFromLive(msg: msg)
    }
}
```

**关键决策**：步骤 6 的 await 让 LiveStore.pauseForCall 等到 CallStore 接听完成（包括 publish + joinRtc）。如果 CallStore 失败（state 切到 .ended），CallStoreObserver 会立刻触发 resumeCall。

**自检点**：
- 真机：开播 → mock 来电 → 控制台依次见 heartbeat paused / monitor paused / liveAgora leave / WS .calling / CallStore.connecting
- 异常路径：mock CallStore.acceptIncomingFromLive 失败 → callState 仍 1（依赖 observer 触发 resumeCall 复位）

**估时**：0.4d

#### T3 — LiveStore.resumeCall() + rejoinLiveChannel() + 15s 倒计时（0.6d）

**文件**：`Sources/Live/LiveStore.swift`

**改动点**：

```swift
extension LiveStore {
    /// CallStore 通话结束（state→.ended）后触发，启动 15s 倒计时回直播（对齐 H5 liveRoom.vue:220）。
    /// 倒计时期间用户端列表显示主播状态仍是 CALLING（H5 getOnlineStatus checkOnCall 为 true）。
    func resumeCall() async {
        guard callState == 1 else { return }
        guard !isWaitingReturnLive else { return }  // 防重复触发
        
        print("🟢 [LiveStore] resumeCall 启动 15s 倒计时")
        
        isWaitingReturnLive = true
        returnLiveCountdown = 15
        
        // 取消可能的旧 task
        returnLiveTask?.cancel()
        returnLiveTask = Task { @MainActor [weak self] in
            guard let self else { return }
            // 15s 倒计时（每秒 -1，UI 显示用）
            for _ in 0..<15 {
                try? await Task.sleep(nanoseconds: 1_000_000_000)
                if Task.isCancelled { return }
                if self.returnLiveCountdown > 0 { self.returnLiveCountdown -= 1 }
            }
            
            await self.rejoinLiveChannel()
            self.heartbeat.start()    // 复用 B 里程碑 start/stop；start 内会先 stop 重置 failureCount
            self.monitor.start()
            self.callState = 0
            self.isWaitingReturnLive = false
            WSHeartbeat.shared.notifyCallStateChanged(callState: 0)
            print("✅ [LiveStore] 已恢复直播")
        }
    }
    
    /// 用 B 里程碑保留的 channelId/rtcToken 重新 join 直播频道。
    /// rtcToken 超过 12h 或 nil 时主动续期。
    private func rejoinLiveChannel() async {
        guard let channelId = agoraChannelId,
              let uid = SessionStore.shared.user?.userId else {
            print("⚠️ [LiveStore] rejoinLiveChannel 缺 channelId/uid")
            return
        }
        
        // 校验 token age（防止暂停期间 token 过期）
        var token = rtcToken ?? ""
        if token.isEmpty {
            do {
                let tokenRes = try await LiveService.getAgoraRtmToken()
                token = tokenRes.rtcToken ?? ""
                // 写回 LiveStore.rtcToken（B 里程碑需暴露 setter，或加内部更新方法）
            } catch {
                print("⚠️ [LiveStore] 续 rtcToken 失败 err=\(error.localizedDescription)")
                return
            }
        }
        guard !token.isEmpty else { return }
        
        liveAgora.join(channelId: channelId,
                       token: token,
                       uid: UInt(uid),
                       profile: .liveBroadcasting)
    }
}
```

**自检点**：
- 通话挂断后 15s 倒计时 15→0 在 UI 上能看到（M3 任务实现 UI）
- rejoinLiveChannel 成功后控制台见 `🟢 [Agora] 本地加入成功`
- 倒计时期间用户主动下播 → returnLiveTask 被 endLive() cancel，不再触发 rejoin

**估时**：0.6d
